import javax.swing.*; import java.awt.*; import java.awt.event.*; import java.util.ArrayList; class Medicine { private String name; private String description; private String category; private int stock; public Medicine(String name, String description, String category, int stock) { this.name = name; this.description = description; this.category = category; this.stock = stock; } public String getName() { return name; } public String getDescription() { return description; } public String getCategory() { return category; } public int getStock() { return stock; } public void addStock(int qty) { stock += qty; } public void sell(int qty) throws Exception { if (qty > stock) { throw new Exception("Not enough stock!"); } stock -= qty; } } public class PharmacySystem { static ArrayList medicines = new ArrayList<>(); public static void main(String[] args) { JFrame frame = new JFrame("Pharmacy Inventory System"); frame.setSize(700, 550); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setLocationRelativeTo(null); JPanel panel = new JPanel(new BorderLayout(10,10)); panel.setBorder(BorderFactory.createEmptyBorder(10,10,10,10)); JLabel title = new JLabel("PHARMACY INVENTORY SYSTEM", JLabel.CENTER); title.setFont(new Font("Arial", Font.BOLD, 22)); title.setForeground(new Color(0,102,204)); panel.add(title, BorderLayout.NORTH); JPanel form = new JPanel(new GridLayout(7,2,10,10)); JTextField nameField = new JTextField(); JTextField descField = new JTextField(); JTextField qtyField = new JTextField(); String[] categories = {"Capsule", "Tablet", "Syrup"}; JComboBox categoryBox = new JComboBox<>(categories); JButton addBtn = new JButton("Add Medicine"); JButton sellBtn = new JButton("Sell Medicine"); JButton deleteBtn = new JButton("Delete Medicine"); JButton viewBtn = new JButton("View Inventory"); form.add(new JLabel("Medicine Name:")); form.add(nameField); form.add(new JLabel("Description:")); form.add(descField); form.add(new JLabel("Category:")); form.add(categoryBox); form.add(new JLabel("Quantity:")); form.add(qtyField); form.add(addBtn); form.add(sellBtn); form.add(deleteBtn); form.add(viewBtn); panel.add(form, BorderLayout.CENTER); JTextArea outputArea = new JTextArea(12,20); outputArea.setEditable(false); outputArea.setFont(new Font("Monospaced", Font.PLAIN, 14)); panel.add(new JScrollPane(outputArea), BorderLayout.SOUTH); frame.add(panel); frame.setVisible(true); // ========================== // ADD // ========================== addBtn.addActionListener(e -> { try { String name = nameField.getText(); String desc = descField.getText(); String category = (String) categoryBox.getSelectedItem(); int qty = Integer.parseInt(qtyField.getText()); if(name.isEmpty() || desc.isEmpty()){ throw new Exception("All fields required!"); } medicines.add(new Medicine(name, desc, category, qty)); outputArea.setText("Medicine added successfully!"); } catch (NumberFormatException ex) { outputArea.setText("Quantity must be a number!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // ========================== // SELL // ========================== sellBtn.addActionListener(e -> { try { String name = nameField.getText(); int qty = Integer.parseInt(qtyField.getText()); for (Medicine m : medicines) { if (m.getName().equalsIgnoreCase(name)) { m.sell(qty); outputArea.setText("Medicine sold successfully!"); return; } } outputArea.setText("Medicine not found!"); } catch (NumberFormatException ex) { outputArea.setText("Quantity must be a number!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // ========================== // DELETE // ========================== deleteBtn.addActionListener(e -> { String name = nameField.getText(); boolean found = false; for (int i = 0; i < medicines.size(); i++) { if (medicines.get(i).getName().equalsIgnoreCase(name)) { medicines.remove(i); found = true; break; } } if (found) { outputArea.setText("Medicine deleted successfully!"); } else { outputArea.setText("Medicine not found!"); } }); // ========================== // VIEW INVENTORY // ========================== viewBtn.addActionListener(e -> { StringBuilder result = new StringBuilder(); int totalStock = 0; result.append("=== INVENTORY LIST ===\n\n"); for (Medicine m : medicines) { result.append("Name: ").append(m.getName()).append("\n"); result.append("Description: ").append(m.getDescription()).append("\n"); result.append("Category: ").append(m.getCategory()).append("\n"); result.append("Stock: ").append(m.getStock()).append("\n"); result.append("---------------------------\n"); totalStock += m.getStock(); } result.append("\nTOTAL STOCK: ").append(totalStock); outputArea.setText(result.toString()); }); } }
