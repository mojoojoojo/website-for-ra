import javax.swing.*; import java.awt.*; import java.awt.event.*; import java.util.ArrayList; // ============================= // Medicine Class // ============================= class Medicine { private String name; private String description; private int stock; private double price; public Medicine(String name, int stock) { this.name = name; this.stock = stock; this.description = generateDescription(name); this.price = generatePrice(name); } private String generateDescription(String name) { switch (name.toLowerCase()) { case "paracetamol": return "Used for fever and mild pain."; case "amoxicillin": return "Antibiotic for bacterial infections."; case "cetirizine": return "Used for allergies and itching."; case "ibuprofen": return "Pain reliever and anti-inflammatory."; default: return "No description available."; } } private double generatePrice(String name) { switch (name.toLowerCase()) { case "paracetamol": return 5.00; case "amoxicillin": return 10.00; case "cetirizine": return 8.00; case "ibuprofen": return 7.00; default: return 6.00; } } public String getName() { return name; } public String getDescription() { return description; } public int getStock() { return stock; } public double getPrice() { return price; } public void addStock(int qty) { stock += qty; } public void sell(int qty) throws Exception { if (qty > stock) { throw new Exception("Not enough stock!"); } stock -= qty; } } // ============================= // Main Pharmacy System // ============================= public class PharmacySystem { static ArrayList medicines = new ArrayList<>(); public static void main(String[] args) { JFrame frame = new JFrame("Pharmacy Inventory System"); frame.setSize(700, 600); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setLocationRelativeTo(null); JPanel panel = new JPanel(new BorderLayout(10,10)); panel.setBorder(BorderFactory.createEmptyBorder(10,10,10,10)); JLabel title = new JLabel("PHARMACY INVENTORY SYSTEM", JLabel.CENTER); title.setFont(new Font("Arial", Font.BOLD, 22)); title.setForeground(new Color(0,102,204)); panel.add(title, BorderLayout.NORTH); // ================= FORM ================= JPanel form = new JPanel(new GridLayout(6,2,10,10)); JTextField nameField = new JTextField(); JTextField ageField = new JTextField(); JTextField medicineField = new JTextField(); JTextField qtyField = new JTextField(); JButton addBtn = new JButton("Add Stock"); JButton sellBtn = new JButton("Sell Medicine"); JButton clearBtn = new JButton("Clear Fields"); JButton viewBtn = new JButton("View Inventory"); form.add(new JLabel("Name:")); form.add(nameField); form.add(new JLabel("Age:")); form.add(ageField); form.add(new JLabel("Medicine:")); form.add(medicineField); form.add(new JLabel("Quantity:")); form.add(qtyField); form.add(addBtn); form.add(sellBtn); form.add(clearBtn); form.add(viewBtn); panel.add(form, BorderLayout.CENTER); // ================= OUTPUT ================= JTextArea outputArea = new JTextArea(15,20); outputArea.setEditable(false); outputArea.setFont(new Font("Monospaced", Font.PLAIN, 14)); panel.add(new JScrollPane(outputArea), BorderLayout.SOUTH); frame.add(panel); frame.setVisible(true); // ================= ADD STOCK ================= addBtn.addActionListener(e -> { try { String medName = medicineField.getText(); int qty = Integer.parseInt(qtyField.getText()); if(medName.isEmpty()){ throw new Exception("Medicine name required!"); } medicines.add(new Medicine(medName, qty)); outputArea.setText("Stock added successfully!"); } catch (NumberFormatException ex) { outputArea.setText("Quantity must be a number!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // ================= SELL + RECEIPT ================= sellBtn.addActionListener(e -> { try { String customerName = nameField.getText(); int age = Integer.parseInt(ageField.getText()); String medName = medicineField.getText(); int qty = Integer.parseInt(qtyField.getText()); for (Medicine m : medicines) { if (m.getName().equalsIgnoreCase(medName)) { m.sell(qty); double total = m.getPrice() * qty; String receipt = "=========== RECEIPT ===========\n" + "Customer Name: " + customerName + "\n" + "Age: " + age + "\n" + "--------------------------------\n" + "Medicine: " + m.getName() + "\n" + "Description: " + m.getDescription() + "\n" + "Price per Unit: ₱" + m.getPrice() + "\n" + "Quantity: " + qty + "\n" + "--------------------------------\n" + "TOTAL: ₱" + total + "\n" + "================================\n" + " Thank you for your purchase!\n"; outputArea.setText(receipt); return; } } outputArea.setText("Medicine not found!"); } catch (NumberFormatException ex) { outputArea.setText("Age and Quantity must be numbers!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // ================= CLEAR ================= clearBtn.addActionListener(e -> { nameField.setText(""); ageField.setText(""); medicineField.setText(""); qtyField.setText(""); outputArea.setText("Fields cleared."); }); // ================= VIEW INVENTORY ================= viewBtn.addActionListener(e -> { StringBuilder result = new StringBuilder(); int totalStock = 0; result.append("====== INVENTORY LIST ======\n\n"); for (Medicine m : medicines) { result.append("Medicine: ").append(m.getName()).append("\n"); result.append("Description: ").append(m.getDescription()).append("\n"); result.append("Stock: ").append(m.getStock()).append("\n"); result.append("Price: ₱").append(m.getPrice()).append("\n"); result.append("------------------------------\n"); totalStock += m.getStock(); } result.append("\nTOTAL STOCK: ").append(totalStock); outputArea.setText(result.toString()); }); } }
