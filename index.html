import javax.swing.*; import java.awt.*; import java.awt.event.*; class Pharmacy { // Instance variables for stock int paracetamolStock = 100; int amoxicillinStock = 100; int cetirizineStock = 100; int ibuprofenStock = 100; // Sell medicine method public String sellMedicine(String medicine, int qty) throws Exception { String description; double price; switch (medicine.toLowerCase()) { case "paracetamol": description = "Used for fever and mild pain."; price = 5.0; if(qty > paracetamolStock) throw new Exception("Not enough stock!"); paracetamolStock -= qty; break; case "amoxicillin": description = "Antibiotic for bacterial infections."; price = 10.0; if(qty > amoxicillinStock) throw new Exception("Not enough stock!"); amoxicillinStock -= qty; break; case "cetirizine": description = "Used for allergies."; price = 8.0; if(qty > cetirizineStock) throw new Exception("Not enough stock!"); cetirizineStock -= qty; break; case "ibuprofen": description = "Pain reliever and anti-inflammatory."; price = 7.0; if(qty > ibuprofenStock) throw new Exception("Not enough stock!"); ibuprofenStock -= qty; break; default: throw new Exception("Medicine not available!"); } return "Description: " + description + "\nPrice per Unit: ₱" + price; } // View current stock public String viewStock() { return "=== CURRENT STOCK ===\n" + "Paracetamol: " + paracetamolStock + "\n" + "Amoxicillin: " + amoxicillinStock + "\n" + "Cetirizine: " + cetirizineStock + "\n" + "Ibuprofen: " + ibuprofenStock + "\n"; } // Get price for receipt public double getPrice(String medicine) { switch (medicine.toLowerCase()) { case "paracetamol": return 5.0; case "amoxicillin": return 10.0; case "cetirizine": return 8.0; case "ibuprofen": return 7.0; default: return 0; } } } public class PharmacySystem { public static void main(String[] args) { JFrame frame = new JFrame("Pharmacy System"); frame.setSize(700,600); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setLocationRelativeTo(null); Pharmacy pharmacy = new Pharmacy(); // Instance of Pharmacy JPanel panel = new JPanel(new BorderLayout(10,10)); panel.setBorder(BorderFactory.createEmptyBorder(10,10,10,10)); JLabel title = new JLabel("PHARMACY INVENTORY SYSTEM", JLabel.CENTER); title.setFont(new Font("Arial", Font.BOLD, 22)); title.setForeground(new Color(0,102,204)); panel.add(title, BorderLayout.NORTH); // Form JPanel form = new JPanel(new GridLayout(6,2,10,10)); JTextField nameField = new JTextField(); JTextField ageField = new JTextField(); JTextField medicineField = new JTextField(); JTextField qtyField = new JTextField(); JButton sellBtn = new JButton("Sell Medicine"); JButton clearBtn = new JButton("Clear Fields"); JButton viewBtn = new JButton("View Stock"); form.add(new JLabel("Name:")); form.add(nameField); form.add(new JLabel("Age:")); form.add(ageField); form.add(new JLabel("Medicine:")); form.add(medicineField); form.add(new JLabel("Quantity:")); form.add(qtyField); form.add(sellBtn); form.add(clearBtn); form.add(viewBtn); panel.add(form, BorderLayout.CENTER); JTextArea outputArea = new JTextArea(15,20); outputArea.setEditable(false); outputArea.setFont(new Font("Monospaced", Font.PLAIN, 14)); panel.add(new JScrollPane(outputArea), BorderLayout.SOUTH); frame.add(panel); frame.setVisible(true); // SELL BUTTON sellBtn.addActionListener(e -> { try { String customerName = nameField.getText(); int age = Integer.parseInt(ageField.getText()); String medicine = medicineField.getText(); int qty = Integer.parseInt(qtyField.getText()); String medInfo = pharmacy.sellMedicine(medicine, qty); double total = pharmacy.getPrice(medicine) * qty; String receipt = "=========== RECEIPT ===========\n" + "Customer Name: " + customerName + "\n" + "Age: " + age + "\n" + "--------------------------------\n" + "Medicine: " + medicine.toUpperCase() + "\n" + medInfo + "\n" + "Quantity: " + qty + "\n" + "--------------------------------\n" + "TOTAL: ₱" + total + "\n" + "================================\n" + "Thank you for your purchase!\n"; outputArea.setText(receipt); } catch (NumberFormatException ex) { outputArea.setText("Age and Quantity must be numbers!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // CLEAR BUTTON clearBtn.addActionListener(e -> { nameField.setText(""); ageField.setText(""); medicineField.setText(""); qtyField.setText(""); outputArea.setText("Fields cleared."); }); // VIEW STOCK viewBtn.addActionListener(e -> { outputArea.setText(pharmacy.viewStock()); }); } }
