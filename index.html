import javax.swing.*; import java.awt.*; import java.awt.event.*; import java.util.ArrayList; // ======================= // Encapsulation // ======================= class Medicine { private String name; private int stock; public Medicine(String name, int stock) { this.name = name; this.stock = stock; } public String getName() { return name; } public int getStock() { return stock; } public void addStock(int qty) { stock += qty; } public void sell(int qty) throws Exception { if (qty > stock) { throw new Exception("Not enough stock!"); } stock -= qty; } } // ======================= // Abstraction // ======================= abstract class Person { protected String name; abstract void role(); } // ======================= // Inheritance + Polymorphism // ======================= class Pharmacist extends Person { public Pharmacist(String name) { this.name = name; } @Override void role() { System.out.println(name + " is managing the pharmacy."); } } // ======================= // Main System Class // ======================= public class PharmacySystem { static ArrayList medicines = new ArrayList<>(); public static void main(String[] args) { // Sample Data medicines.add(new Medicine("Paracetamol", 20)); medicines.add(new Medicine("Biogesic", 15)); medicines.add(new Medicine("Amoxicillin", 10)); // GUI Frame JFrame frame = new JFrame("Pharmacy Inventory System"); frame.setSize(600, 450); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setLocationRelativeTo(null); // Main Panel JPanel mainPanel = new JPanel(); mainPanel.setLayout(new BorderLayout(10,10)); mainPanel.setBorder(BorderFactory.createEmptyBorder(10,10,10,10)); // Title JLabel title = new JLabel("PHARMACY INVENTORY SYSTEM", JLabel.CENTER); title.setFont(new Font("Arial", Font.BOLD, 22)); title.setForeground(new Color(0,102,204)); mainPanel.add(title, BorderLayout.NORTH); // Center Panel (Form) JPanel formPanel = new JPanel(); formPanel.setLayout(new GridLayout(4,2,10,10)); JTextField nameField = new JTextField(); JTextField qtyField = new JTextField(); JButton addBtn = new JButton("Add Medicine"); JButton sellBtn = new JButton("Sell Medicine"); JButton viewBtn = new JButton("View Inventory"); formPanel.add(new JLabel("Medicine Name:")); formPanel.add(nameField); formPanel.add(new JLabel("Quantity:")); formPanel.add(qtyField); formPanel.add(addBtn); formPanel.add(sellBtn); formPanel.add(viewBtn); mainPanel.add(formPanel, BorderLayout.CENTER); // Output Area JTextArea outputArea = new JTextArea(8,20); outputArea.setEditable(false); outputArea.setFont(new Font("Monospaced", Font.PLAIN, 14)); JScrollPane scrollPane = new JScrollPane(outputArea); mainPanel.add(scrollPane, BorderLayout.SOUTH); frame.add(mainPanel); frame.setVisible(true); // ========================== // Button Functionalities // ========================== // Add Medicine addBtn.addActionListener(e -> { try { String name = nameField.getText(); int qty = Integer.parseInt(qtyField.getText()); if(name.isEmpty()){ throw new Exception("Medicine name required!"); } medicines.add(new Medicine(name, qty)); outputArea.setText("Medicine added successfully!"); } catch (NumberFormatException ex) { outputArea.setText("Quantity must be a number!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // Sell Medicine sellBtn.addActionListener(e -> { try { String name = nameField.getText(); int qty = Integer.parseInt(qtyField.getText()); for (Medicine m : medicines) { if (m.getName().equalsIgnoreCase(name)) { m.sell(qty); outputArea.setText("Medicine sold successfully!"); return; } } outputArea.setText("Medicine not found!"); } catch (NumberFormatException ex) { outputArea.setText("Quantity must be a number!"); } catch (Exception ex) { outputArea.setText(ex.getMessage()); } }); // View Inventory viewBtn.addActionListener(e -> { StringBuilder result = new StringBuilder(); result.append("=== INVENTORY LIST ===\n"); for (Medicine m : medicines) { result.append(m.getName()) .append(" - Stock: ") .append(m.getStock()) .append("\n"); } outputArea.setText(result.toString()); }); } }
